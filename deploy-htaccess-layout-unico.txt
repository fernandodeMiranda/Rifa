# Use este arquivo SOMENTE se voce optar pelo layout de pasta unica
# (tudo dentro de htdocs/, sem pastas irmas fora do document root).
#
# Renomeie este arquivo para ".htaccess" e coloque na raiz de htdocs/,
# no lugar dos dois arquivos ".htaccess" separados (o de public/ e o da
# raiz do projeto) - eles nao podem coexistir na mesma pasta.

RewriteEngine On

# Bloqueia acesso direto ao .env e a pastas internas da aplicacao,
# caso elas fiquem dentro do document root publico.
<Files ".env">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
    <IfModule !mod_authz_core.c>
        Order deny,allow
        Deny from all
    </IfModule>
</Files>

RewriteRule ^(app|database|scripts|storage)/ - [F,L]

# Serve arquivos e diretorios existentes diretamente (assets, uploads).
RewriteCond %{REQUEST_FILENAME} -f [OR]
RewriteCond %{REQUEST_FILENAME} -d
RewriteRule ^ - [L]

# Tudo o mais passa pelo front controller.
RewriteRule ^ index.php [L]
